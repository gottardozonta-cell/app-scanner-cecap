function normalizeGoogleSheetUrl(raw){ if(!raw) return ''; const value = raw.trim(); if(!value) return ''; const sheetIdMatch = value.match(/\/spreadsheets\/d\/([A-Za-z0-9-_]+)/i) || value.match(/[?&]id=([A-Za-z0-9-_]+)/i); const fileIdMatch = value.match(/\/file\/d\/([A-Za-z0-9-_]+)/i) || value.match(/(?:\/d\/|id=)([A-Za-z0-9-_]{10,})/i); const foundId = sheetIdMatch && sheetIdMatch[1] ? sheetIdMatch[1] : (fileIdMatch && fileIdMatch[1] ? fileIdMatch[1] : null); if(foundId){ let url = 'https://docs.google.com/spreadsheets/d/' + foundId + '/export?format=csv'; const gid = value.match(/[?&]gid=([0-9]+)/i) || value.match(/#gid=([0-9]+)/i); if(gid && gid[1]) url += '&gid=' + gid[1]; return url; } if(/^https?:\/\//i.test(value) && /\.csv($|\?)/i.test(value)) return value; return value; } function normalizeQtyValue(value){ if(value === null || value === undefined) return 0; const s = String(value).trim(); if(!s) return 0; const cleaned = s.replace(/\./g, '').replace(',', '.').replace(/[^0-9.\-]/g, ''); if(!cleaned) return 0; const num = Number.parseFloat(cleaned); return Number.isFinite(num) ? num : 0; }
